  \section{Research}
  \begin{onecolentry}
    \textbf{Building an Agent Based Model to Study the Environmental Adaptability of Sessile/Motile State Switching Strategies in \textit{Bacillus subtilis}}

    \textit{Advisor: \href{https://cqb.pku.edu.cn/tanglab/index.htm}{Prof. Chao Tang}}
    \begin{highlights}
      \item Built an agent based model from steady environment measurements to simulate sessile and motile state switching in fluctuating chemical environments.
      \item Used transfer entropy to compare how agents with bidirectional regulation and one way regulation use environmental information in decision making.
      \item Evaluated the adaptive value of the two switching strategies during chemotaxis.
    \end{highlights}

    \vspace{0.12cm}
    \textbf{Using a Function Driven Optimization Model to Study the Coding Principles of the \textit{Drosophila} Olfactory Circuit in the Dual Tasks of Gas Motion Sensing and Gradient Sensing}

    \textit{Advisors: \href{https://emonet.biology.yale.edu/}{Prof. Thierry Emonet} and \href{https://kschen.scholar.princeton.edu/bio}{Dr. Kevin Chen}}
    \begin{highlights}
      \item Extended a circuit for odor motion sensing into a parameterized network that can be optimized.
      \item Designed a joint task for odor motion and odor gradient sensing under a metabolic constraint and optimized network parameters for task performance.
      \item Analyzed the tradeoff between the two sensory tasks within a shared network.
    \end{highlights}

    \vspace{0.12cm}
    \textbf{Dynamics and Renormalization Analysis of the Emergence of Low Rank Structure in Artificial Neural Networks}

    \textit{Advisors: \href{https://users.flatironinstitute.org/~ytu10/}{Prof. Yuhai Tu} and Dr. Yikuan Zhang}
    \begin{highlights}
      \item Trained multilayer perceptrons on MNIST and recorded representations and singular value spectra throughout learning.
      \item Studied the stages of low rank structure formation by varying network width and learning rate and by regularizing intermediate singular values.
      \item Examined whether the observed scale dependence admits a useful renormalization description.
    \end{highlights}
  \end{onecolentry}
